# a) Nos piden que definamos una clase "Persona", la cual tenga como atributos el nombre, 
# el apellido la fecha de nacimiento, la dirección, el teléfono y el email de una persona.
# b) Además deberemos declarar el método age() dentro de la clase que nos devuelva la edad de la
# persona calculándolo con la fecha actual (pista: debemos importar el módulo 'datetime')
# c) Modifiquemos la clase agregando un atributo edad que se calcule cuando se instancie la clase

##Solución: 
#a)
import datetime # esto lo usamos para el cálculo de fechas

class Person:

    def __init__(self, nombre, apellido, fechaNac, direccion, telefono, email):
        self.nombre = nombre
        self.apellido = apellido
        self.fechaNac = fechaNac

        self.direccion = direccion
        self.telefono = telefono
        self.email = email

# b)
    def age(self):
        hoy = datetime.date.today()
        edad = hoy.year - self.fechaNac.year

        if hoy < datetime.date(hoy.year, self.fechaNac.month, self.fechaNac.day):
            edad -= 1

        return edad
